```dataviewJS
TABLE WITHOUT ID
    tag AS "导数定义",
    rows.file.link AS "01第一章",
    rows.section.text AS "题目预览"
```

```dataview
TABLE WITHOUT ID
  choice(
    meta(section).subpath, 
    link(file.link, "第" + regexreplace(meta(section).subpath, "\D", "") + "题"),
    file.link
  ) AS "题目位置",
  regexreplace(
    regexreplace(section.text, "^(.*?)(#连续.*?)(---|$)", "$2"),
    "---.*", 
    ""
  ) AS "题目预览"
FROM "武忠祥强化"
FLATTEN file.sections AS section
WHERE contains(section.text, "#连续") 
  AND section.text != ""
GROUP BY file.link
```
